{
  read(expr);
  read(state);

  work = [[0, expr], 0];
  size = 1;
  vals = 0;

  while (size > 0) {
    task = work[0];
    work = work[1];
    size -= 1;

    if (task[0] == 0) {
      node = task[1];

      if (node[0] == 0)
        vals = [node[1], vals]
      elif (node[0] == 1) {
        name = node[1];
        cur = state;
        found = 0;
        value = 0;
        while (cur[0] == 1 && found == 0) {
          pair = cur[1];
          if (pair[0] == name) {
            value = pair[1];
            found = 1
          }
          cur = cur[2]
        }
        vals = [value, vals]
      }
      else {
        work = [[1, node[1]], work];
        work = [[0, node[3]], work];
        work = [[0, node[2]], work];
        size += 3
      }
    }
    else {
      op = task[1];
      right = vals[0];
      vals = vals[1];
      left = vals[0];
      vals = vals[1];

      if (op == "+") res = left + right
      elif (op == "-") res = left - right
      elif (op == "*") res = left * right
      elif (op == "/") res = left / right
      elif (op == "%") res = left % right
      elif (op == "==") res = left == right
      elif (op == "!=") res = left != right
      elif (op == "<") res = left < right
      elif (op == "<=") res = left <= right
      elif (op == ">") res = left > right
      elif (op == ">=") res = left >= right
      elif (op == "&&") res = left && right
      else res = left !! right;

      vals = [res, vals]
    }
  }

  write(vals[0])
}
